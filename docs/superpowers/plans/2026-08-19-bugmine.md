> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 从已采集的 trace 中确定性挖掘三类 AI bug 信号（test_failure / fix_commit / user_correction），落 `bug_events` 表，在 dashboard 新增 Bugs 聚合 tab。

**Tech Stack:** Python 3.11+, sqlite3（PRAGMA user_version 迁移）, aiohttp web, 单文件 dashboard HTML/JS, pytest。

**Spec:** `docs/superpowers/specs/2026-08-19-bugmine-design.md`

```python
    def _create_v7_tables(self, conn: sqlite3.Connection) -> None:
        conn.execute(
            """
            CREATE TABLE IF NOT EXISTS bug_events (
                id INTEGER PRIMARY KEY,
                session_id TEXT NOT NULL,
                ts TEXT NOT NULL,
                client TEXT NOT NULL DEFAULT '',
                model TEXT NOT NULL DEFAULT '',
                cwd TEXT NOT NULL DEFAULT '',
                signal_type TEXT NOT NULL,
                title TEXT NOT NULL DEFAULT '',
                excerpt TEXT NOT NULL DEFAULT '',
                record_index INTEGER NOT NULL DEFAULT 0,
                healed INTEGER,
                FOREIGN KEY (session_id) REFERENCES sessions(id) ON DELETE CASCADE
            )
            """
        )
        conn.execute("CREATE INDEX IF NOT EXISTS idx_bug_events_ts ON bug_events(ts)")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_bug_events_signal ON bug_events(signal_type)")
        conn.execute(
            """
            CREATE TABLE IF NOT EXISTS bugmine_sources (
                session_id TEXT PRIMARY KEY,
                processed_at TEXT NOT NULL
            )
            """
        )

    def _migrate_v6_to_v7(self, conn: sqlite3.Connection) -> None:
        self._create_v7_tables(conn)
        conn.execute(f"PRAGMA user_version = {SCHEMA_VERSION}")
        conn.commit()
```

6. 存取方法（放在 `delete_session` 附近，遵循现有 `with self._connect() as conn:` 风格）。注意 trace_store.py 的写入方法用 `_write_access` 上下文管理器（见 `create_session`/:116）——写方法（record_bug_events/mark_bugmine_processed）用 `_write_access`，读方法用 `_connect`，以文件现有机制为准；`datetime`/`timedelta`/`timezone` 如未 import 需补：

```python
    def record_bug_events(self, session_id: str, events: list[dict]) -> None:
        """Insert mined bug events for a session (healed: True/False/None → 1/0/NULL)."""
        with self._connect() as conn:
            conn.executemany(
                """INSERT INTO bug_events
                   (session_id, ts, client, model, cwd, signal_type, title, excerpt, record_index, healed)
                   VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)""",
                [
                    (
                        session_id,
                        ev["ts"],
                        ev["client"],
                        ev["model"],
                        ev["cwd"],
                        ev["signal_type"],
                        ev["title"],
                        ev["excerpt"],
                        ev["record_index"],
                        None if ev["healed"] is None else int(bool(ev["healed"])),
                    )
                    for ev in events
                ],
            )

    def is_bugmine_processed(self, session_id: str) -> bool:
        with self._connect() as conn:
            row = conn.execute(
                "SELECT 1 FROM bugmine_sources WHERE session_id = ?", (session_id,)
            ).fetchone()
        return row is not None

    def mark_bugmine_processed(self, session_id: str, processed_at: str) -> None:
        with self._connect() as conn:
            conn.execute(
                "INSERT OR REPLACE INTO bugmine_sources (session_id, processed_at) VALUES (?, ?)",
                (session_id, processed_at),
            )

    def bug_stats(self, days: int) -> dict:
        """Pre-aggregated stats for the Bugs dashboard tab (server-side GROUP BY)."""
        since = (datetime.now(timezone.utc) - timedelta(days=days)).isoformat()
        with self._connect() as conn:
            by_signal = [
                dict(r)
                for r in conn.execute(
                    "SELECT signal_type, COUNT(*) AS count FROM bug_events WHERE ts >= ? GROUP BY signal_type ORDER BY count DESC",
                    (since,),
                )
            ]
            trend = [
                dict(r)
                for r in conn.execute(
                    "SELECT substr(ts, 1, 10) AS day, signal_type, COUNT(*) AS count FROM bug_events WHERE ts >= ? GROUP BY day, signal_type ORDER BY day",
                    (since,),
                )
            ]
            by_project = [
                dict(r)
                for r in conn.execute(
                    "SELECT cwd, COUNT(*) AS count FROM bug_events WHERE ts >= ? GROUP BY cwd ORDER BY count DESC",
                    (since,),
                )
            ]
            by_client_model = [
                dict(r)
                for r in conn.execute(
                    "SELECT client, model, COUNT(*) AS count FROM bug_events WHERE ts >= ? GROUP BY client, model ORDER BY count DESC",
                    (since,),
                )
            ]
            heal = conn.execute(
                "SELECT COUNT(*) AS total, COALESCE(SUM(healed), 0) AS healed FROM bug_events WHERE signal_type = 'test_failure' AND ts >= ?",
                (since,),
            ).fetchone()
            mined = [
                dict(r)
                for r in conn.execute(
                    "SELECT s.client AS client, COUNT(*) AS sessions FROM bugmine_sources b JOIN sessions s ON s.id = b.session_id GROUP BY s.client"
                )
            ]
        return {
            "by_signal": by_signal,
            "trend": trend,
            "by_project": by_project,
            "by_client_model": by_client_model,
            "heal_rate": {"total": int(heal["total"]), "healed": int(heal["healed"])},
            "mined_sessions_by_client": mined,
        }

    def bug_events_list(
        self, *, days: int, signal_type: str | None, cwd: str | None, limit: int
    ) -> list[dict]:
        since = (datetime.now(timezone.utc) - timedelta(days=days)).isoformat()
        sql = "SELECT * FROM bug_events WHERE ts >= ?"
        params: list[object] = [since]
        if signal_type:
            sql += " AND signal_type = ?"
            params.append(signal_type)
        if cwd:
            sql += " AND cwd = ?"
            params.append(cwd)
        sql += " ORDER BY ts DESC LIMIT ?"
        params.append(limit)
        with self._connect() as conn:
            return [dict(r) for r in conn.execute(sql, params)]
```

```python
from claude_tap.bugmine.detectors import detect_session


def _req_record(ts, messages):
    return {"timestamp": ts, "request": {"body": {"messages": messages}}, "response": {"body": {}}}


def _bash_use(uid, command):
    return {"role": "assistant", "content": [{"type": "tool_use", "id": uid, "name": "Bash", "input": {"command": command}}]}


def _tool_result(uid, text, is_error=False):
    block = {"type": "tool_result", "tool_use_id": uid, "content": text}
    if is_error:
        block["is_error"] = True
    return {"role": "user", "content": [block]}


def _user_text(text):
    return {"role": "user", "content": [{"type": "text", "text": text}]}


def test_test_failure_and_heal():
    records = [
        _req_record("t1", [_bash_use("u1", "pytest tests/")]),
        _req_record("t2", [_bash_use("u1", "pytest tests/"), _tool_result("u1", "FAILED tests/test_a.py::test_x - assert 1 == 2", True)]),
```

```python
"""Deterministic bug-signal detectors over a session's trace records.

v1 supports the Anthropic message format only (request.body.messages content
blocks + response.body.content). Records of other clients yield zero events.
"""

from __future__ import annotations

import hashlib
import re
from dataclasses import dataclass
from typing import Any, Iterator

from claude_tap.prompt_kb.messages import _keep_text

_TEST_FAILURE_RE = re.compile(r"(\d+ failed|FAILED|ERROR|Traceback)", re.IGNORECASE)
_TEST_PASSED_RE = re.compile(r"\d+ passed", re.IGNORECASE)
_GIT_COMMIT_RE = re.compile(r"\bgit\s+(?:-C\s+\S+\s+)?commit\b")
_COMMIT_MSG_RE = re.compile(r"-m\s+(['\"])(?P<msg>.*?)\1", re.DOTALL)
_FIX_PREFIX_RE = re.compile(r"^fix[(:/\s]", re.IGNORECASE)
_FIX_IN_CMD_RE = re.compile(r"\bfix[(:/\s]", re.IGNORECASE)
_CORRECTION_RE = re.compile(
    r"(不对|错了|搞错|有 ?bug|不对劲|wrong|incorrect|broken|that's not)", re.IGNORECASE
)

_TITLE_CHARS = 120
_EXCERPT_CHARS = 300
_CORRECTION_TITLE_CHARS = 60


@dataclass
class BugEvent:
    signal_type: str
    record_index: int
    ts: str
    title: str
    excerpt: str
    healed: bool | None = None


def _crop(text: str, max_chars: int) -> str:
    text = text.strip()
    return text if len(text) <= max_chars else text[: max_chars - 1] + "…"


def _blocks(record: dict[str, Any]) -> Iterator[tuple[str, dict[str, Any]]]:
    """Yield (role, content_block) from both request history and response body."""
    request = record.get("request")
    body = request.get("body") if isinstance(request, dict) else None
    if not isinstance(body, dict):
        return
    messages = body.get("messages")
    if not isinstance(messages, list):
        return  # non-Anthropic format: unsupported in v1
    for msg in messages:
        if not isinstance(msg, dict):
            continue
        content = msg.get("content")
        if not isinstance(content, list):
            continue
        for block in content:
            if isinstance(block, dict):
                yield str(msg.get("role") or ""), block
    response = record.get("response")
    resp_body = response.get("body") if isinstance(response, dict) else None
    if isinstance(resp_body, dict):
        content = resp_body.get("content")
        if isinstance(content, list):
            for block in content:
                if isinstance(block, dict):
                    yield "assistant", block


def _result_text(block: dict[str, Any]) -> str:
    content = block.get("content")
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        return "\n".join(
            b["text"] for b in content if isinstance(b, dict) and isinstance(b.get("text"), str)
        )
    return ""


def _first_failure_line(text: str) -> str:
    for line in text.splitlines():
        if _TEST_FAILURE_RE.search(line):
            return line.strip()
    return text.strip()


def _fix_commit_event(command: str, record_index: int, ts: str) -> BugEvent | None:
    if not _GIT_COMMIT_RE.search(command):
        return None
    match = _COMMIT_MSG_RE.search(command)
    if match:
        first_line = match.group("msg").strip().splitlines()[0].strip()
        if not _FIX_PREFIX_RE.match(first_line):
            return None
        title = first_line
    else:
        if not _FIX_IN_CMD_RE.search(command):
            return None
        title = command.strip()[:_TITLE_CHARS]
    return BugEvent("fix_commit", record_index, ts, _crop(title, _TITLE_CHARS), _crop(command, _EXCERPT_CHARS))


def detect_session(records: list[dict[str, Any]]) -> list[BugEvent]:
    """Scan one session's records; dedup by tool_use_id / content hash across records."""
    events: list[BugEvent] = []
    seen_uses: set[str] = set()
    seen_results: set[str] = set()
    seen_texts: set[str] = set()
    pending_bash: dict[str, str] = {}
    failures: list[tuple[int, BugEvent]] = []
    passed_seqs: list[int] = []
    seq = 0
    for rec_idx, record in enumerate(records):
        if not isinstance(record, dict):
            continue
        ts = str(record.get("timestamp") or "")
        for role, block in _blocks(record):
            seq += 1
            btype = block.get("type")
            if btype == "tool_use" and block.get("name") == "Bash":
                uid = str(block.get("id") or "")
                if not uid or uid in seen_uses:
                    continue
                seen_uses.add(uid)
                command = str((block.get("input") or {}).get("command") or "")
                pending_bash[uid] = command
                event = _fix_commit_event(command, rec_idx, ts)
                if event is not None:
                    events.append(event)
            elif btype == "tool_result":
                uid = str(block.get("tool_use_id") or "")
                if not uid or uid in seen_results:
                    continue
                seen_results.add(uid)
                text = _result_text(block)
                if block.get("is_error"):
                    if _TEST_FAILURE_RE.search(text):
                        event = BugEvent(
                            "test_failure",
                            rec_idx,
                            ts,
                            _crop(pending_bash.get(uid, "") or "(unknown command)", _TITLE_CHARS),
                            _crop(_first_failure_line(text), _EXCERPT_CHARS),
                        )
                        failures.append((seq, event))
                        events.append(event)
                elif _TEST_PASSED_RE.search(text):
                    passed_seqs.append(seq)
            elif btype == "text" and role == "user":
                text = str(block.get("text") or "").strip()
                if not text or not _keep_text(text) or not _CORRECTION_RE.search(text):
                    continue
                digest = hashlib.sha256(text.encode()).hexdigest()
                if digest in seen_texts:
                    continue
                seen_texts.add(digest)
                events.append(
                    BugEvent(
                        "user_correction",
                        rec_idx,
                        ts,
                        _crop(text, _CORRECTION_TITLE_CHARS),
                        _crop(text, _EXCERPT_CHARS),
                    )
                )
    for fseq, event in failures:
        event.healed = any(p > fseq for p in passed_seqs)
    return events
```

```python
"""Incremental bug-event mining over trace sessions (mirrors prompt_kb.extract)."""

from __future__ import annotations

import logging
from datetime import datetime, timezone
from typing import Any

from claude_tap.bugmine.detectors import detect_session
from claude_tap.trace_store import TraceStore

logger = logging.getLogger(__name__)


def _now() -> str:
    return datetime.now(timezone.utc).isoformat()


def _record_model(record: dict[str, Any]) -> str:
    request = record.get("request") if isinstance(record.get("request"), dict) else {}
    body = request.get("body") if isinstance(request.get("body"), dict) else {}
    return str(body.get("model") or "")


def mine_unprocessed(trace: TraceStore, *, limit: int = 50) -> dict[str, int]:
    """Mine recent unmined sessions; failures stay unmarked and retry next pass."""
    processed = events_total = skipped = 0
    rows = trace.list_session_rows(limit=limit * 4)  # over-fetch: some are filtered below
    for row in rows:
        if processed >= limit:
            break
        session_id = str(row["id"])
        if trace.is_bugmine_processed(session_id):
            continue
        if not row["record_count"]:
            trace.mark_bugmine_processed(session_id, _now())
            skipped += 1
            continue
        try:
            records = trace.load_records(session_id)
            cwd = str(row["cwd"] if "cwd" in row.keys() else (trace.load_session_row(session_id) or {}).get("cwd") or "")
            client = str(row["client"] or "unknown")
            events = []
            for ev in detect_session(records):
                record = records[ev.record_index] if ev.record_index < len(records) else {}
                events.append(
                    {
                        "signal_type": ev.signal_type,
                        "ts": ev.ts or _now(),
                        "client": client,
                        "model": _record_model(record),
                        "cwd": cwd,
                        "title": ev.title,
                        "excerpt": ev.excerpt,
                        "record_index": ev.record_index,
                        "healed": ev.healed,
                    }
                )
            if events:
                trace.record_bug_events(session_id, events)
            trace.mark_bugmine_processed(session_id, _now())
        except Exception as exc:  # noqa: BLE001 - not marked, retried next round
            logger.warning("bugmine failed for session %s: %s", session_id, exc)
            skipped += 1
            continue
        processed += 1
        events_total += len(events)
    return {"processed": processed, "events": events_total, "skipped": skipped}
```

注：`row["cwd"]` 是否可用取决于 `list_session_rows` 的 SELECT 列——实现时先看 trace_store.py 里该方法；若有 cwd 就直接用，删掉 fallback。`load_records` 返回的是 payload dict 列表（`extract.py:111` 同款用法）。

```js
async function loadBugs() {
  const [statsResp, eventsResp] = await Promise.all([
    fetch("/api/bugs/stats?days=30"),
    fetch("/api/bugs/events?days=30&limit=100"),
  ]);
  const stats = await statsResp.json();

  renderBugsEvents(eventsPayload.events || []);
}

function renderBugsStats(stats) {
  $("#bugs-by-signal").innerHTML = (stats.by_signal || [])
    .map(r => `<div class="bugs-row"><span>${escapeHtml(r.signal_type)}</span><b>${r.count}</b></div>`).join("");
  $("#bugs-trend").innerHTML = (stats.trend || [])
    .map(r => `<div class="bugs-row"><span>${escapeHtml(r.day)}</span><span>${escapeHtml(r.signal_type)}</span><b>${r.count}</b></div>`).join("");
  $("#bugs-by-project").innerHTML = (stats.by_project || [])
    .map(r => `<div class="bugs-row"><span title="${escapeHtml(r.cwd)}">${escapeHtml(r.cwd || "(unknown)")}</span><b>${r.count}</b></div>").join("");
  $("#bugs-by-client-model").innerHTML = (stats.by_client_model || [])
    .map(r => `<div class="bugs-row"><span>${escapeHtml(r.client)} / ${escapeHtml(r.model)}</span><b>${r.count}</b></div>").join("");
  const heal = stats.heal_rate || {total: 0, healed: 0};
  const pct = heal.total ? Math.round((heal.healed / heal.total) * 100) : 0;
  $("#bugs-heal-rate").innerHTML = `<div class="bugs-row"><b>${pct}%</b><span>${heal.healed}/${heal.total}</span></div>`;
}

function renderBugsEvents(events) {
  const body = $("#bugs-events-body");
  $("#bugs-empty").hidden = events.length > 0;
  body.innerHTML = events.map(ev => `
    <tr data-session="${escapeHtml(ev.session_id)}" class="bugs-event-row">
      <td>${escapeHtml((ev.ts || "").slice(0, 16).replace("T", " "))}</td>
      <td>${escapeHtml(ev.signal_type)}</td>
      <td>${escapeHtml(ev.title)}</td>
      <td title="${escapeHtml(ev.cwd)}">${escapeHtml(ev.cwd || "")}</td>
      <td>${escapeHtml(ev.client)}</td>
      <td>${ev.healed === null || ev.healed === undefined ? "—" : ev.healed ? "✓" : "✗"}</td>
    </tr>`).join("");
  body.querySelectorAll(".bugs-event-row").forEach(row => {
    row.addEventListener("click", () => {
      const sid = row.getAttribute("data-session");
      if (sid) location.href = `/dashboard/session/${encodeURIComponent(sid)}`;
    });
  });
}
```

注意：`escapeHtml` 与 `$` 以 dashboard.html 里现有的同名工具为准，没有就用现有等价物；`hidden` 属性的切换要照现有 view 切换代码对 KB 面板的处理方式（可能用的是 class 或 style.display，随现有机制，不强求 `hidden`）。

```python
"""`claude-tap bugs` subcommands."""

from __future__ import annotations

import json
import sys

from claude_tap.bugmine.mine import mine_unprocessed
from claude_tap.trace_store import TraceStore


def bugs_main(argv: list[str]) -> int:
    if not argv or argv[0] != "mine":
        print("usage: claude-tap bugs mine [--limit N]", file=sys.stderr)
        return 2
    limit = 50
    if "--limit" in argv:
        idx = argv.index("--limit")
        try:
            limit = int(argv[idx + 1])
        except (IndexError, ValueError):
            print("usage: claude-tap bugs mine [--limit N]", file=sys.stderr)
            return 2
    result = mine_unprocessed(TraceStore.default(), limit=limit)
    print(json.dumps(result, ensure_ascii=False))
    return 0
```
